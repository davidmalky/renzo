// FEATURE FUNCTIONS - PRIORITY SCORING function calculatePriorityScore(contact) { let score = 0; const today = Date.now(); // Days since last contact (40% weight) if (contact.lastContactDate) { const daysSince = Math.floor((today - contact.lastContactDate) / (1000 * 60 * 60 * 24)); const daysFactor = Math.min(daysSince / 90, 1) * 100; score += daysFactor * 0.4; } // Contract expiry (30% weight) if (contact.contractExpiry) { const daysUntil = Math.floor((contact.contractExpiry - today) / (1000 * 60 * 60 * 24)); if (daysUntil >= 0 && daysUntil <= 90) { const expiryFactor = ((90 - daysUntil) / 90) * 100; score += expiryFactor * 0.3; } } // Invoice amount (30% weight) if (contact.invoiceAmount && contact.invoiceAmount > 0) { const invoiceFactor = Math.min((contact.invoiceAmount / 10000) * 100, 100); score += invoiceFactor * 0.3; } return Math.round(score); } function computeOutreachTrigger(contact) { const today = Date.now(); // Check contract expiry first if (contact.contractExpiry) { const daysUntil = Math.floor((contact.contractExpiry - today) / (1000 * 60 * 60 * 24)); if (daysUntil >= 0 && daysUntil <= 90) { return `Contract expires in ${daysUntil} day${daysUntil === 1 ? '' : 's'}`; } } // Check invoice amount if (contact.invoiceAmount && contact.invoiceAmount > 0) { const formatted = contact.invoiceAmount.toLocaleString('en-US', { style: 'currency', currency: 'USD', maximumFractionDigits: 0 }); return `Invoice ${formatted} overdue`; } // Check last contact date if (contact.lastContactDate) { const daysSince = Math.floor((today - contact.lastContactDate) / (1000 * 60 * 60 * 24)); if (daysSince > 30) { return `Not contacted in ${daysSince} days`; } } return null; } function getPriorityBadgeColor(score) { if (score >= 80) return 'priority-critical'; if (score >= 60) return 'priority-high'; if (score >= 40) return 'priority-medium'; return 'priority-low'; } function sortContactsByPriority(contacts) { return contacts.map(c => ({ ...c, priorityScore: calculatePriorityScore(c) })).sort((a, b) => b.priorityScore - a.priorityScore); } // BATCH OUTREACH function parseNaturalLanguageQuery(query) { const lower = query.toLowerCase(); if (lower.includes('contract') && (lower.includes('expir') || lower.includes('end'))) { const match = lower.match(/(\d+)\s*day/); const days = match ? parseInt(match[1]) : 90; return { type: 'contractExpiry', operator: 'within', value: days }; } if (lower.includes('invoice')) { const match = lower.match(/\$?(\d+)/); const amount = match ? parseInt(match[1]) : 0; return { type: 'invoice', operator: 'above', value: amount }; } if (lower.includes('not contacted') || lower.includes('haven\'t contacted')) { const match = lower.match(/(\d+)\s*day/); const days = match ?
